================================================================================
PARTE 10 — PRIMEIRO DEPLOY (CLONAR DE NOVO)
================================================================================

mkdir -p /opt/apps/scraper
cd /opt/apps/scraper

git clone -b development https://github.com/MoesioFiuza/Scrapper-Unificado.git repo

cd repo

# Python
python3 -m venv venv
./venv/bin/pip install -r requirements.txt

# PostgreSQL (utilizador e base)
sudo -u postgres psql -c "CREATE USER scraper WITH PASSWORD 'SUA_SENHA';" 2>/dev/null || true
sudo -u postgres psql -c "CREATE DATABASE scraper_unificado OWNER scraper;" 2>/dev/null || true

# .env
cp deploy/env.production.example .env
nano .env   # SECRET_KEY, DATABASE_URL, etc.

# Chrome headless (scraping)
apt update
apt install -y google-chrome-stable chromium-chromedriver

# Frontend — build no PC e push, OU no servidor se tiver node:
cd frontend && npm ci && npm run build && cd ..

# Migrar users.json legado (se existir)
./venv/bin/python scripts/migrate_json_to_db.py

# Systemd
cp deploy/scraper.service /etc/systemd/system/scraper.service
systemctl daemon-reload
systemctl enable scraper
systemctl start scraper

# Nginx
cp deploy/nginx/scraper.conf /etc/nginx/sites-available/scraper
ln -sf /etc/nginx/sites-available/scraper /etc/nginx/sites-enabled/scraper
nginx -t && systemctl reload nginx

# Testar
curl http://127.0.0.1:8001/api/health
curl http://212.47.68.222/scraper/api/health
